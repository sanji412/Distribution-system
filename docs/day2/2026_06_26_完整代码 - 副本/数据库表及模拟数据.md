### 一、数据库创建（分库）

```sql
-- 创建各微服务独立数据库
CREATE DATABASE IF NOT EXISTS user_db DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE DATABASE IF NOT EXISTS product_db DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE DATABASE IF NOT EXISTS stock_db DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE DATABASE IF NOT EXISTS order_db DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE DATABASE IF NOT EXISTS pay_db DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
```

### 二、各库表结构 + 索引 + Seata undo_log（仅事务参与库）

#### 1. 用户中心库（user_db）

```sql
USE user_db;

-- 用户表
DROP TABLE IF EXISTS sys_user;
CREATE TABLE sys_user (
    user_id BIGINT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(100) NOT NULL,
    nickname VARCHAR(50),
    role VARCHAR(20) NOT NULL,
    status TINYINT DEFAULT 1,
    phone VARCHAR(20),
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 登录场景唯一索引
ALTER TABLE sys_user ADD UNIQUE INDEX idx_username (username);
```

#### 2. 商品中心库（product_db）

```sql
USE product_db;

-- 商品表
DROP TABLE IF EXISTS product;
CREATE TABLE product (
    product_id BIGINT PRIMARY KEY AUTO_INCREMENT,
    product_name VARCHAR(100) NOT NULL,
    category VARCHAR(50),
    price DECIMAL(10,2) NOT NULL,
    sku_code VARCHAR(50) UNIQUE,
    description VARCHAR(255),
    status TINYINT DEFAULT 1,
    safe_stock INT DEFAULT 20,
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 浏览历史表（AI推荐数据源）
DROP TABLE IF EXISTS browse_history;
CREATE TABLE browse_history (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id BIGINT NOT NULL,
    product_id BIGINT NOT NULL,
    browse_time DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 浏览记录查询索引
ALTER TABLE browse_history ADD INDEX idx_user_browse (user_id, browse_time DESC);
```

#### 3. 库存中心库（stock_db）+ Seata undo_log

```sql
USE stock_db;

-- 仓库表
DROP TABLE IF EXISTS warehouse;
CREATE TABLE warehouse (
    warehouse_id BIGINT PRIMARY KEY AUTO_INCREMENT,
    warehouse_name VARCHAR(50) NOT NULL,
    warehouse_address VARCHAR(200),
    status TINYINT DEFAULT 1,
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 库存表
DROP TABLE IF EXISTS stock;
CREATE TABLE stock (
    stock_id BIGINT PRIMARY KEY AUTO_INCREMENT,
    product_id BIGINT NOT NULL,
    warehouse_id BIGINT NOT NULL,
    stock_num INT NOT NULL,
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 库存唯一索引（单商品单仓库唯一）
ALTER TABLE stock ADD UNIQUE INDEX idx_product_warehouse (product_id, warehouse_id);

-- Seata AT模式回滚日志表（事务参与库必须）
DROP TABLE IF EXISTS undo_log;
CREATE TABLE undo_log (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    branch_id BIGINT NOT NULL,
    xid VARCHAR(100) NOT NULL,
    context VARCHAR(128) NOT NULL,
    rollback_info LONGBLOB NOT NULL,
    log_status INT NOT NULL,
    log_created DATETIME NOT NULL,
    log_modified DATETIME NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

### 四、使用说明

1. **执行顺序**：先执行「数据库创建」→ 再执行「各库表结构」→ 最后执行「模拟数据」；
2. **Seata 适配**：undo_log 表仅在 stock_db/order_db/pay_db 创建（事务参与库），符合 Seata AT 模式规范；
3. 演示场景覆盖：
   - 库存不足场景：罗技 M720 鼠标库存 3 件，订单 DD202505190004 下单 2 件触发库存不足回滚；
   - 订单全状态：已完成 / 已支付 / 待支付 / 库存不足 / 已发货 5 种核心状态；
   - AI 客服：覆盖物流查询、搭配推荐、库存预警、热销趋势 4 类对话；
   - 分布式事务：订单 - 库存 - 支付链路的事务提交 / 回滚场景；
4. **字符集**：统一使用 utf8mb4 适配中文 + 特殊字符，避免乱码；
5. **密码说明**：用户密码为 123456（BCrypt 加密），可直接用于登录演示。

    password VARCHAR(100) NOT NULL COMMENT '登录密码',
    nickname VARCHAR(50) COMMENT '用户昵称',
    role VARCHAR(20) NOT NULL COMMENT '用户角色：admin-管理员、user-普通用户',
    status TINYINT DEFAULT 1 COMMENT '账号状态：0-禁用、1-启用',
    phone VARCHAR(20) COMMENT '联系电话',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
) COMMENT = '系统用户表';

-- 2.2 商品中心 - 商品表
CREATE TABLE product (
    product_id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '商品主键ID',
    product_name VARCHAR(100) NOT NULL COMMENT '商品名称',
    category VARCHAR(50) COMMENT '商品分类',
    price DECIMAL(10,2) NOT NULL COMMENT '商品单价',
    sku_code VARCHAR(50) UNIQUE COMMENT '商品SKU编码',
    description VARCHAR(255) COMMENT '商品描述',
    status TINYINT DEFAULT 1 COMMENT '上下架状态：0-下架、1-上架',
    safe_stock INT DEFAULT 20 COMMENT '安全库存阈值，用于库存预警',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
) COMMENT = '商品表';

-- 2.3 商品中心 - 用户浏览历史表
CREATE TABLE browse_history (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键ID',
    user_id BIGINT NOT NULL COMMENT '浏览用户ID',
    product_id BIGINT NOT NULL COMMENT '被浏览的商品ID',
    browse_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '浏览时间'
) COMMENT = '用户浏览历史表';

-- 2.4 库存中心 - 仓库表
CREATE TABLE warehouse (
    warehouse_id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '仓库主键ID',
    warehouse_name VARCHAR(50) NOT NULL COMMENT '仓库名称',
    warehouse_address VARCHAR(200) COMMENT '仓库详细地址',
    status TINYINT DEFAULT 1 COMMENT '状态：0-停用、1-启用',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
) COMMENT = '仓库表';

-- 2.5 库存中心 - 库存表
CREATE TABLE stock (
    stock_id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '库存记录ID',
    product_id BIGINT NOT NULL COMMENT '关联商品ID',
    warehouse_id BIGINT NOT NULL COMMENT '所属仓库ID',
    stock_num INT NOT NULL COMMENT '当前可用库存数量',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
) COMMENT = '库存表';

-- 2.6 订单中心 - 订单主表
CREATE TABLE order_main (
    order_id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '订单主键ID',
    order_no VARCHAR(32) NOT NULL UNIQUE COMMENT '订单编号（格式：DD + 日期 + 序号）',
    user_id BIGINT NOT NULL COMMENT '下单用户ID',
    product_name VARCHAR(100) NOT NULL COMMENT '商品名称（冗余）',
    product_num INT NOT NULL COMMENT '购买数量（冗余）',
    total_amount DECIMAL(10,2) NOT NULL COMMENT '订单总金额',
    order_status VARCHAR(20) NOT NULL COMMENT '订单状态：待支付、已支付、已发货、已完成、库存不足',
    receiver_name VARCHAR(50) COMMENT '收货人姓名',
    receiver_phone VARCHAR(20) COMMENT '收货人电话',
    receiver_address VARCHAR(200) COMMENT '收货地址',
    logistics_company VARCHAR(50) COMMENT '物流公司',
    logistics_no VARCHAR(50) COMMENT '物流单号',
    logistics_status VARCHAR(20) COMMENT '物流状态：待发货、运输中、已签收',
    current_location VARCHAR(100) COMMENT '当前物流位置',
    expect_arrive_time DATETIME COMMENT '预计送达时间',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '下单时间',
    pay_time DATETIME COMMENT '支付时间',
    delivery_time DATETIME COMMENT '发货时间',
    finish_time DATETIME COMMENT '订单完成时间'
) COMMENT = '订单主表';

-- 2.7 订单中心 - 订单明细表
CREATE TABLE order_item (
    item_id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '明细主键ID',
    order_id BIGINT NOT NULL COMMENT '关联订单主表ID',
    product_id BIGINT NOT NULL COMMENT '商品ID',
    product_name VARCHAR(100) NOT NULL COMMENT '商品名称（冗余）',
    product_price DECIMAL(10,2) NOT NULL COMMENT '商品单价（冗余）',
    quantity INT NOT NULL COMMENT '购买数量',
    sub_total DECIMAL(10,2) NOT NULL COMMENT '明细小计金额',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
) COMMENT = '订单明细表';

-- 2.8 订单中心 - AI对话记录表
CREATE TABLE ai_chat_record (
    chat_id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '对话记录ID',
    user_id BIGINT NOT NULL COMMENT '提问用户ID',
    question TEXT NOT NULL COMMENT '用户提问内容',
    answer TEXT NOT NULL COMMENT 'AI回复内容',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '对话时间'
) COMMENT = 'AI对话记录表';

-- 2.9 支付中心 - 支付单表
CREATE TABLE payment (
    pay_id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '支付单主键ID',
    pay_no VARCHAR(32) NOT NULL UNIQUE COMMENT '支付单号',
    order_no VARCHAR(32) NOT NULL COMMENT '关联订单编号',
    user_id BIGINT NOT NULL COMMENT '支付用户ID',
    pay_amount DECIMAL(10,2) NOT NULL COMMENT '支付金额',
    pay_method VARCHAR(20) COMMENT '支付方式',
    pay_status VARCHAR(20) NOT NULL COMMENT '支付状态：待支付、支付成功、支付失败',
    callback_content TEXT COMMENT '支付回调报文内容',
    callback_time DATETIME COMMENT '支付回调时间',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
) COMMENT = '支付单表';

-- 2.10 Seata分布式事务回滚表（Day1单体阶段预留，适配后续演进）
CREATE TABLE undo_log (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键ID',
    branch_id BIGINT NOT NULL COMMENT '分支事务ID',
    xid VARCHAR(100) NOT NULL COMMENT '全局事务ID',
    context VARCHAR(128) NOT NULL COMMENT '事务上下文信息',
    rollback_info LONGBLOB NOT NULL COMMENT '回滚日志数据（数据修改前镜像）',
    log_status INT NOT NULL COMMENT '日志状态',
    log_created DATETIME NOT NULL COMMENT '日志创建时间',
    log_modified DATETIME NOT NULL COMMENT '日志修改时间'
) COMMENT = 'Seata AT模式事务回滚日志表';

-- 3. 索引创建（优化高频查询）

CREATE INDEX idx_sys_user_username ON sys_user(username);

CREATE INDEX idx_browse_history_user_time ON browse_history(user_id, browse_time DESC);

-- 库存表索引
CREATE UNIQUE INDEX uk_stock_product_warehouse ON stock(product_id, warehouse_id);

-- 订单主表索引
CREATE INDEX idx_order_main_user_id ON order_main(user_id);
CREATE INDEX idx_order_main_status ON order_main(order_status);
CREATE INDEX idx_order_main_create_time ON order_main(create_time);

-- 订单明细表索引
CREATE INDEX idx_order_item_order_id ON order_item(order_id);
CREATE INDEX idx_order_item_product_id ON order_item(product_id);

-- 支付单表索引
CREATE INDEX idx_payment_order_no ON payment(order_no);

-- AI对话记录表索引
CREATE INDEX idx_ai_chat_create_time ON ai_chat_record(create_time);
